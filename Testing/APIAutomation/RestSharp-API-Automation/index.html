    1. Send a GET Request and Validate Status Code

    Explanation:
    RestSharp allows sending requests and validating responses in a few lines.

    var client = new RestClient("https://api.example.com");
var request = new RestRequest("users", Method.Get);
var response = client.Execute(request);
Assert.AreEqual(200, (int)response.StatusCode);

    2. Send a POST Request to Create a New User

    Explanation:
    Sends a JSON body with user data and checks if the response returns a 201 Created status.

    var client = new RestClient("https://api.example.com");
var request = new RestRequest("users", Method.Post);
request.AddJsonBody(new { name = "John", job = "Engineer" });
var response = client.Execute(request);
Assert.AreEqual(201, (int)response.StatusCode);

    3. Validate a JSON Response Field

    Explanation:
    Validates the email field in the JSON response.

    var content = JObject.Parse(response.Content);
Assert.AreEqual("john@example.com", content["data"]["email"]);

    4. Add Headers in a Request

    Explanation:
    Adds Content-Type and Authorization headers.

    request.AddHeader("Content-Type", "application/json");
request.AddHeader("Authorization", "Bearer your_token");

    5. Validate API Response Time

    Explanation:
    Checks if the response time is less than 2 seconds.

    Assert.LessOrEqual(response.ResponseTime.TotalMilliseconds, 2000);

    6. Verify JSON Field Presence

    Explanation:
    Checks if the response JSON has a "data" key.

    var content = JObject.Parse(response.Content);
Assert.IsTrue(content.ContainsKey("data"));

    7. Send a PUT Request

    Explanation:
    Sends a PUT request to update user information.

    var request = new RestRequest("users/2", Method.Put);
request.AddJsonBody(new { name = "John", job = "Manager" });
var response = client.Execute(request);
Assert.AreEqual(200, (int)response.StatusCode);

    8. Handle Pagination

    Explanation:
    Keeps requesting pages until there are no more records.

    int page = 1;
while (true)
{
    var request = new RestRequest("users", Method.Get).AddParameter("page", page++);
    var response = client.Execute(request);
    var data = JObject.Parse(response.Content)["data"];
    if (!data.HasValues) break;
    Console.WriteLine(data);
}

    9. Verify JSON Array Size

    Explanation:
    Checks if the data array has exactly six items.

    var data = JArray.Parse(JObject.Parse(response.Content)["data"].ToString());
Assert.AreEqual(6, data.Count);

    10. Extract Specific Data from JSON

    Explanation:
    Uses JSON parsing to extract specific fields.

    var email = JObject.Parse(response.Content)["data"]["email"].ToString();

    11. Send a DELETE Request

    Explanation:
    Removes a user resource, expecting status 204 No Content.

    var request = new RestRequest("users/2", Method.Delete);
var response = client.Execute(request);
Assert.AreEqual(204, (int)response.StatusCode);

    12. Validate Specific Header in Response

    Explanation:
    Checks if Content-Type header matches expected value.

    Assert.AreEqual("application/json; charset=utf-8", response.ContentType);

    13. Handle Query Parameters

    Explanation:
    Passes query parameters like page=2.

    request.AddParameter("page", 2);

    14. Handle Basic Authentication

    Explanation:
    Provides username and password for basic auth.

    client.Authenticator = new HttpBasicAuthenticator("username", "password");

    15. Send a PATCH Request for Partial Update

    Explanation:
    Updates only specific fields without overwriting all data.

    var request = new RestRequest("users/2", Method.Patch);
request.AddJsonBody(new { job = "Architect" });
var response = client.Execute(request);
Assert.AreEqual(200, (int)response.StatusCode);

    16. Add Multiple Query Parameters

    Explanation:
    Adds multiple query parameters using repeated AddParameter() calls.

    request.AddParameter("page", 1);
request.AddParameter("limit", 10);

    17. Check for Non-Null JSON Path Value

    Explanation:
    Ensures data.email is not null.

    var email = JObject.Parse(response.Content)["data"]["email"];
Assert.IsNotNull(email);

    18. Send Form Parameters

    Explanation:
    Used when content type is application/x-www-form-urlencoded.

    request.AddHeader("Content-Type", "application/x-www-form-urlencoded");
request.AddParameter("username", "john");
request.AddParameter("password", "1234");
